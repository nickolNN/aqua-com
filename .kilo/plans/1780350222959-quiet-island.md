### 1.2 Fix `useEffect` side effect in FilterBar.tsx

**File:** `src/components/layout/FilterBar.tsx` (lines 34-36)

## Phase 2: Medium — Type Safety & Performance

### 2.1 Fix `any` type in Button.tsx

**File:** `src/components/ui/Button.tsx` (line 13)

**Current:**
```ts
(({ className, variant = 'default', size = 'default', as, ...rest }: any, ref) => { ... });
```

**Fix:** Use proper type intersection:
```ts
const Button = React.forwardRef<HTMLButtonElement | HTMLAnchorElement, ButtonProps & React.AnchorHTMLAttributes<HTMLAnchorElement>>(
  ({ className, variant = 'default', size = 'default', as, ...rest }, ref) => {
    const variantStyles: Record<string, string> = { ... };
    const sizeStyles: Record<string, string> = { ... };
    const buttonClasses = cn(
      'inline-flex items-center justify-center rounded-md text-sm font-medium transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-offset-2 disabled:pointer-events-none disabled:opacity-50',
      variantStyles[variant],
      sizeStyles[size],
      className
    );

    if (as === 'a') {
      const anchorProps = rest as React.AnchorHTMLAttributes<HTMLAnchorElement>;
      return React.createElement('a', { ref: ref as any, className: buttonClasses, ...anchorProps, ...rest });
    }

    return React.createElement('button', { ref, className: buttonClasses, ...rest });
  }
);
```

### 3.3 Fix localStorage initialization in settingsSlice.ts

**File:** `src/stores/settingsSlice.ts` (lines 14-16)

**Fix:** Wrap in try-catch and add SSR-safe check:
```ts
const getThemeFromStorage = (): 'light' | 'dark' => {
  try {
    const stored = localStorage.getItem('theme');
    if (stored === 'light' || stored === 'dark') return stored;
  } catch (e) {
    console.warn('localStorage access denied', e);
  }
  return getSystemThemePreference();
};

const initialState: SettingsState = {
```

## Phase 4: Redux Improvements

### 4.1 Fix tasksSlice direct mutation

**File:** `src/stores/tasksSlice.ts` (lines 47-52)

**Current:**
```ts
updateTaskStatus: (state, action) => {
  const task = state.tasks.find((t) => t.id === taskId);
  if (task) { task.status = newStatus as Task['status']; }
},
```

**Fix:** Use immutable update pattern:
```ts
updateTaskStatus: (state, action) => {
  const { taskId, newStatus } = action.payload;
  state.tasks = state.tasks.map((t) =>
    t.id === taskId ? { ...t, status: newStatus as Task['status'] } : t
  );
},
```
